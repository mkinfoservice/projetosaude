# assets/uploads/.htaccess
# Proteção completa da pasta de uploads

# Desabilitar listagem de diretório
Options -Indexes

# Bloquear execução de scripts PHP/outras linguagens
<FilesMatch "\.(php|php\d|phtml|phar|php3|php4|php5|php7|php8|pl|py|jsp|asp|sh|cgi)$">
    Order Deny,Allow
    Deny from all
</FilesMatch>

# Forçar download de arquivos (segurança extra)
<FilesMatch "\.(pdf|doc|docx|xls|xlsx|jpg|jpeg|png|gif)$">
    Header set Content-Disposition attachment
</FilesMatch>

# Headers de segurança
<IfModule mod_headers.c>
    Header set X-Content-Type-Options "nosniff"
    Header set X-Frame-Options "SAMEORIGIN"
</IfModule>

# Cache control para arquivos estáticos
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/jpg "access plus 1 month"
    ExpiresByType image/jpeg "access plus 1 month"
    ExpiresByType image/png "access plus 1 month"
    ExpiresByType application/pdf "access plus 1 week"
</IfModule>

# Bloquear acesso direto a .htaccess
<Files ".htaccess">
    Order Allow,Deny
    Deny from all
</Files>
